SELECT * from employee;

{1. Find employees hired after 1 January 2025.}
SELECT * FROM employee
WHERE hire_date>'2025-01-01';

{2. Find employees hired between two dates.}
SELECT * FROM employee
WHERE hire_date>'2025-01-01' and hire_date>'2026-01-01';

{3. Find employees hired in the current year.}
SELECT * FROM employee
WHERE YEAR(hire_date)=YEAR(curdate());

{4. Find employees hired in the current month.}
SELECT * FROM employee
WHERE month(hire_date)=month(curdate());

{5. Find employees hired in the last 30 days.}
SELECT * FROM employee
WHERE hire_date>=curdate()-INTERVAL 30 day;

{6. Find the number of employees hired in each year.}
SELECT YEAR(hire_date), COUNT(*) AS total_employee FROM employee
GROUP BY YEAR(hire_date);

{7. Find the latest hired employee.}
SELECT * FROM employee
ORDER BY hire_date DESC limit 1;

{8. Find the second latest hired employee.}
SELECT * FROM employee
WHERE hire_date=(SELECT max(hire_date) FROM employee
WHERE hire_date<(SELECT max(hire_date) FROM employee));

{9. Find employees hired on weekends.}
SELECT * FROM employee
WHERE dayofweek(hire_date) IN (1,7);

{10. Find department-wise latest hire date.}
SELECT department_id, MAX(hire_date) AS laste_employee
FROM employee
GROUP BY department_id;
