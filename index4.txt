            manager_id INT NULL,

SELECT * FROM employee;

{1. Display 'High' if salary > 80,000 otherwise 'Low'.}
SELECT * ,
CASE
	WHEN salary>80000 THEN 'HIGH'
    ELSE
		'LOW'
	END AS salary_per
FROM employee;

{2. Display salary grades using CASE.}
SELECT first_name, last_name, salary,
CASE
	WHEN salary>=100000 THEN 'A'
    WHEN salary>=80000 AND salary<=99999 then 'B'
    WHEN salary>=60000 AND salary<=79999 THEN 'C'
    WHEN salary>=40000 AND salary<=59999 THEN 'D'
    ELSE
		'C'
	END AS salary_grade
FROM employee;

{3. Replace NULL department_id with 0.}
SELECT *,
	IFNULL(department_id,0) AS department_id
FROM employee;

{4. Display 'Not Available' when phone is NULL.}
SELECT *,
	IFNULL(phone,'NOT AVAILABLE') AS phone
FROM employee;

{5. Count employees with NULL department_id.}
SELECT COUNT(*) AS null_department_id
FROM employee
WHERE department_id is NULL;

{6. Find employees where manager_id is NULL.}
SELECT *
FROM employee
WHERE manager_id is NULL;

{7. Sort employees so NULL salaries appear last}
SELECT * FROM employee
ORDER BY salary DESC;
