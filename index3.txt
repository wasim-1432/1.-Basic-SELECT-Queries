SELECT * FROM employee;

{1. Convert first_name to uppercase.}
SELECT employee_id, UPPER(first_name), last_name, department_id, salary, hire_date
FROM employee;

{2. Convert first_name to lowercase.}
SELECT employee_id, LOWER(first_name), last_name, department_id, salary, hire_date
FROM employee;

{3. Find the length of each employee name.}
SELECT CONCAT(first_name,' ',last_name) AS full_name, LENGTH(full_name) AS full_name_length
FROM employee;

{4. Display first_name and last_name as one full_name.}
SELECT first_name,last_name, concat(first_name,' ',last_name) AS full_name
FROM employee;

5. Find employees whose name length is greater than 5.
SELECT * from employee
WHERE LENGTH(first_name)>5;

{6. Find the first 3 characters of each name.}
SELECT first_name, SUBSTRING(TRIM(first_name),1,3) AS three_character
FROM employee;

{7. Remove leading/trailing spaces from a name.}
SELECT first_name, TRIM(first_name) AS remove_spaces
FROM employee;

{8. Replace a specific character in employee names.}
SELECT first_name, REPLACE(first_name,'e','aa')
FROM employee;

{9. Count employees whose names start with 'A'.}
SELECT COUNT(*) AS employee_count_whose_name_start_A
FROM employee 
WHERE first_name LIKE 'A%';

{10. Find duplicate full names.}
SELECT concat(first_name,' ',last_name), COUNT(*) AS duplacate_name
FROM employee
GROUP BY concat(first_name,' ',last_name) HAVING COUNT(*)>1;
