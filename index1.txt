SELECT * FROM employee;

{1. Find the total number of employees.}
SELECT COUNT(*) AS Total_Employee
FROM employee;

{2. Find the number of non-NULL salaries.}
SELECT COUNT(*) AS Total_Employee
FROM employee
WHERE salary is not NULL;

{3. Find the total salary of all employees.}
SELECT SUM(salary) AS Total_Salary_All_Employees
FROM employee;

{4. Find the average salary.}
SELECT AVG(salary) AS Total_Avg_Salary
FROM employee;

{5. Find the minimum and maximum salary.}
SELECT MIN(salary) AS Minimum_Salary, MAX(salary) AS Maximum_Salary
FROM employee;

{6. Find department-wise employee count.}
SELECT department_id, COUNT(*) AS Employee_Count
FROM employee
GROUP BY department_id;

{7. Find department-wise average salary.}
SELECT department_id, AVG(salary) AS avg_salary
from employee
GROUP BY department_id;

{8. Find department-wise maximum salary.}
SELECT department_id, MAX(salary) AS maximum_salary
FROM employee
GROUP BY department_id;

{9. Find departments having more than 5 employees.}
SELECT department_id, COUNT(*) AS department_more_than_5_emp
FROM employee
GROUP BY department_id having count(*)>5;

{10. Find departments whose average salary is greater than 60,000.}
SELECT department_id, AVG(salary) AS avg_salary
FROM employee
group by department_id having AVG(salary)>60000;

{11. Find the total salary paid by each department.}
SELECT department_id, SUM(salary) AS total_salary_each_dapartment
FROM employee
group by department_id;
