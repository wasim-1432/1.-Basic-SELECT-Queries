SELECT * FROM employee;

{1. Find duplicate email addresses.}
SELECT email, COUNT(*) AS duplicate_email_count
FROM employee
GROUP BY email HAVING COUNT(*)>1;

{2. Find duplicate employee names.}
SELECT CONCAT(first_name,' ',last_name), COUNT(*) AS duplicate_name_count
FROM employee
GROUP BY CONCAT(first_name,' ',last_name) HAVING COUNT(*)>1;

{3. Find departments having duplicate names}
SELECT department_name, COUNT(*) AS duplicate_name
FROM employee
GROUP BY department_name HAVING COUNT(*)>1;

{6. Find records that occur more than once.}
SELECT CONCAT(first_name,' ',last_name) AS duplicate_name, COUNT(*) AS duplicate_count from employee
GROUP BY first_name,last_name HAVING COUNT(*)>1;

{7. Find unique records only.}
SELECT CONCAT(first_name,' ',last_name) AS duplicate_name, COUNT(*) AS duplicate_count from employee
GROUP BY first_name,last_name HAVING COUNT(*)=1;
